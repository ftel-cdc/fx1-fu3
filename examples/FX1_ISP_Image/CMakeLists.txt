cmake_minimum_required(VERSION 3.25)
project(fx1_isp_image C ASM)
get_filename_component(NEON_MDF_PATH "${CMAKE_CURRENT_LIST_DIR}/../../neon-mdf" ABSOLUTE)
include("${NEON_MDF_PATH}/tools/cmake/project.cmake")
if(NOT NEON_MDF_TARGET STREQUAL "FX1" OR NOT TARGET neon_mdf::isp_session_fx1)
  message(FATAL_ERROR "Use FX1 and enable CAMERA_RUNTIME + CAMERA_FX1_BACKEND")
endif()
include("${NEON_MDF_PATH}/cmake/freertos.cmake")
neon_mdf_add_freertos(${NEON_MDF_TARGET} "${CMAKE_CURRENT_SOURCE_DIR}/Inc")
neon_mdf_add_camera_freertos()
neon_mdf_add_executable(fx1_isp_image Src/main.c)
target_link_libraries(fx1_isp_image PRIVATE neon_mdf::camera_freertos)
set(FX1_IMAGE_FRAME_COUNT "8" CACHE STRING "RAW replay frames; four DMA slots rotate")
if(NOT FX1_IMAGE_FRAME_COUNT MATCHES "^[1-9][0-9]*$")
  message(FATAL_ERROR "FX1_IMAGE_FRAME_COUNT must be a positive integer")
endif()
target_compile_definitions(fx1_isp_image PRIVATE FRAME_COUNT=${FX1_IMAGE_FRAME_COUNT})

set(FX1_IMAGE_WIDTH "2688" CACHE STRING "Even replay input width")
set(FX1_IMAGE_HEIGHT "1520" CACHE STRING "Even replay input height")
target_compile_definitions(fx1_isp_image PRIVATE WIDTH=${FX1_IMAGE_WIDTH}u HEIGHT=${FX1_IMAGE_HEIGHT}u)

option(FX1_CAMERA_IRQ_PORT_PROBE "Build the destructive stopped-device IRQ port test (no image)" OFF)
if(FX1_CAMERA_IRQ_PORT_PROBE)
  neon_mdf_add_executable(fx1_camera_irq_port_probe
    "${NEON_MDF_PATH}/tests/camera_hal/fx1_irq_port_probe.c")
  target_link_libraries(fx1_camera_irq_port_probe PRIVATE neon_mdf::isp_session_fx1 neon_mdf::freertos)
endif()
