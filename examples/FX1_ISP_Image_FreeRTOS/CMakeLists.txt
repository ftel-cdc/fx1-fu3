cmake_minimum_required(VERSION 3.25)
project(fx1_isp_image_freertos C ASM)
get_filename_component(NEON_MDF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../neon-mdf" ABSOLUTE)
set(NEON_MDF_ENABLE_FTEL_ISP ON)
include("${NEON_MDF_PATH}/tools/cmake/project.cmake")
if(NOT NEON_MDF_TARGET STREQUAL "FX1")
  message(FATAL_ERROR "Use preset FX1")
endif()
include("${NEON_MDF_PATH}/cmake/freertos.cmake")
# Cấu hình kernel local của example; không phụ thuộc example khác.
neon_mdf_add_freertos(FX1 "${CMAKE_CURRENT_SOURCE_DIR}/Inc")
neon_mdf_add_isp_freertos()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(FX1_ISP_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/profile/profile.csrw" CACHE FILEPATH "Neon application tuning profile")
set(FX1_IMAGE_FRAME_COUNT "" CACHE STRING "Override frame count (1..4); empty uses default 2")
if(NOT FX1_IMAGE_FRAME_COUNT STREQUAL "" AND NOT FX1_IMAGE_FRAME_COUNT MATCHES "^[1-4]$")
  message(FATAL_ERROR "FX1_IMAGE_FRAME_COUNT must be 1..4")
endif()
set(_profile_header "${CMAKE_CURRENT_BINARY_DIR}/generated/profile.h")
add_custom_command(OUTPUT "${_profile_header}"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_profile.py"
    --profile "${FX1_ISP_PROFILE}" --output "${_profile_header}"
  DEPENDS "${FX1_ISP_PROFILE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_profile.py" VERBATIM)
foreach(_target fx1_isp_image_freertos fx1_isp_pipeline_checks)
  neon_mdf_add_executable(${_target} Src/main.c "${_profile_header}")
  target_link_libraries(${_target} PRIVATE neon_mdf::ftel_isp_freertos)
  target_include_directories(${_target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  if(NOT FX1_IMAGE_FRAME_COUNT STREQUAL "")
    target_compile_definitions(${_target} PRIVATE FRAME_COUNT=${FX1_IMAGE_FRAME_COUNT})
  endif()
endforeach()
# Test trì hoãn ACK recovery, tách khỏi ELF ứng dụng và build mặc định.
set_target_properties(fx1_isp_pipeline_checks PROPERTIES EXCLUDE_FROM_ALL TRUE)
target_sources(fx1_isp_pipeline_checks PRIVATE Tests/pipeline_checks.c)
target_include_directories(fx1_isp_pipeline_checks PRIVATE Tests)
target_compile_definitions(fx1_isp_pipeline_checks PRIVATE PIPELINE_CONTRACT_TEST=1)
target_link_options(fx1_isp_pipeline_checks PRIVATE -Wl,--wrap=ftel_isp_fx1_recovery_poll)
