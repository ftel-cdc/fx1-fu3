cmake_minimum_required(VERSION 3.25)
project(fx1_isp_profile_replay C ASM)
get_filename_component(NEON_MDF_PATH "${CMAKE_CURRENT_LIST_DIR}/../../neon-mdf" ABSOLUTE)
include("${NEON_MDF_PATH}/tools/cmake/project.cmake")
if(NOT NEON_MDF_TARGET STREQUAL "FX1")
  message(FATAL_ERROR "Use preset FX1")
endif()
neon_mdf_add_executable(fx1_isp_profile_replay Src/main.c Reference/fx1_isp_drv.c)
target_include_directories(fx1_isp_profile_replay PRIVATE Reference)

option(FX1_PROFILE_BYPASS_BLC_LSC "Bypass BLC and LSC, preserve all other profile writes" OFF)
if(FX1_PROFILE_BYPASS_BLC_LSC)
  target_compile_definitions(fx1_isp_profile_replay PRIVATE FX1_PROFILE_BYPASS_BLC_LSC=1)
endif()

option(FX1_PROFILE_BYPASS_LSC "Bypass only LSC, preserve BLC and all other writes" OFF)
if(FX1_PROFILE_BYPASS_LSC)
  target_compile_definitions(fx1_isp_profile_replay PRIVATE FX1_PROFILE_BYPASS_LSC=1)
endif()
if(FX1_PROFILE_BYPASS_LSC AND FX1_PROFILE_BYPASS_BLC_LSC)
  message(FATAL_ERROR "Choose either LSC-only bypass or BLC+LSC bypass")
endif()

set(FX1_PROFILE_DISABLE_BLOCK NONE CACHE STRING "Disable exactly one block")
set_property(CACHE FX1_PROFILE_DISABLE_BLOCK PROPERTY STRINGS NONE BLC LSC BPC WB DG CCM GAMMA GTM NR_2D EE CNF AEC AWB AF RESIZER)
set(_single_blocks NONE BLC LSC BPC WB DG CCM GAMMA GTM NR_2D EE CNF AEC AWB AF RESIZER)
list(FIND _single_blocks "${FX1_PROFILE_DISABLE_BLOCK}" _single_id)
if(_single_id LESS 0)
  message(FATAL_ERROR "Unknown FX1_PROFILE_DISABLE_BLOCK")
endif()
if(_single_id GREATER 0)
  if(FX1_PROFILE_BYPASS_LSC OR FX1_PROFILE_BYPASS_BLC_LSC)
    message(FATAL_ERROR "Single-block option cannot be combined with older bypass options")
  endif()
  target_compile_definitions(fx1_isp_profile_replay PRIVATE FX1_PROFILE_DISABLE_ID=${_single_id})
endif()
